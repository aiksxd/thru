    22	/// Resolve the shell to spawn: THRU_SHELL > platform default.
    23	/// Unix: $SHELL or /bin/sh; Windows: %COMSPEC% (cmd.exe).
    24	pub fn default_shell() -> String {
    25	    if let Ok(s) = std::env::var("THRU_SHELL") {
    26	        if !s.is_empty() {
    27	            return s;
    28	        }
    29	    }
    30	    if cfg!(unix) {
    31	        std::env::var("SHELL").unwrap_or_else(|_| "/bin/sh".to_string())
    32	    } else {
    33	        // Windows: default to %COMSPEC% (cmd.exe) for predictable script behavior.
    34	        // PowerShell can be selected via THRU_SHELL=powershell.exe.
    35	        std::env::var("COMSPEC").unwrap_or_else(|_| "cmd.exe".to_string())
    36	    }
    37	}

   122	fn parse_size(data: &[u8]) -> io::Result<(u16, u16)> {
   123	    if data.len() < 5 {

   133	/// Interactive PTY shell: SHELL_OPEN then bidirectional raw byte stream.
   134	/// RESIZE frames (op 31) are intercepted; all other frames are keystrokes.
   135	/// Empty frame from client = disconnect.
   136	pub fn handle_shell_open(conn: &mut Box<dyn Connection>, data: &[u8]) -> io::Result<()> {
   137	    let (cols, rows) = parse_size(data)?;
   138	    let mut session = PtySession::new(cols, rows)?;
   139	
   140	    // PTY output -> client frames (own thread).
   141	    let mut out_conn = conn.try_clone()?;
   142	    let mut pty_reader = session
   143	        .take_reader()
   144	        .ok_or_else(|| io::Error::new(io::ErrorKind::Other, "no pty reader"))?;
   145	    let reader_handle = thread::spawn(move || {
   146	        let mut buf = [0u8; 8192];
   147	        loop {
   148	            match pty_reader.read(&mut buf) {
   149	                Ok(0) => break,
   150	                Ok(n) => {
   151	                    if write_frame(&mut out_conn, &buf[..n]).is_err() {
   152	                        break;
   153	                    }
   154	                }
   155	                Err(_) => break,
   156	            }
   157	        }
   158	    });
   159	
   160	    // Client frames -> PTY input (or RESIZE handling).
   161	    let mut pty_writer = session
   162	        .take_writer()
   163	        .ok_or_else(|| io::Error::new(io::ErrorKind::Other, "no pty writer"))?;
   164	    loop {
   165	        if session.try_wait()?.is_some() {
   166	            break;
   167	        }
   168	        match read_frame(conn) {
   169	            Ok(frame) => {
   170	                if frame.is_empty() {
   171	                    break; // client requested disconnect
   172	                }
   173	                if frame[0] == OP_RESIZE {
   174	                    if let Ok((c, r)) = parse_size(&frame) {
   175	                        let _ = session.resize(c, r);
   176	                    }
   177	                } else {
   178	                    pty_writer.write_all(&frame)?;
   179	                    pty_writer.flush()?;
   180	                }
   181	            }
   182	            Err(_) => break,
   183	        }
   184	    }
   185	
   186	    let _ = reader_handle.join();
   187	    Ok(())
   188	}

   190	/// Non-interactive script execution.
   191	/// Uses std::process::Command (not PTY) for reliability:
   192	///   Unix:    <user-shell> -e -c "script"  (set -e = abort on first error; shell from THRU_SHELL/$SHELL)
   193	///   Windows: temp .bat with `if errorlevel 1 exit` after each command
   194	/// Streams stdout+stderr back; final non-empty frame is the i32 exit code.
   195	pub fn handle_exec_script(conn: &mut Box<dyn Connection>, data: &[u8]) -> io::Result<()> {
   196	    if data.len() < 5 {
   197	        return write_frame(conn, &[ST_ERR]);
   198	    }
   199	    let script_len = u32::from_be_bytes([data[1], data[2], data[3], data[4]]) as usize;
   200	    if data.len() < 5 + script_len {
   201	        return write_frame(conn, &[ST_ERR]);
   202	    }
   203	    let script = String::from_utf8_lossy(&data[5..5 + script_len]).to_string();
   204	
   205	    let output = if cfg!(unix) {
   206	        // Use the user's resolved shell (THRU_SHELL > $SHELL > /bin/sh)
   207	        // instead of hardcoding sh, so zsh/bash builtins and aliases work.
   208	        let shell = default_shell();
   209	        Command::new(&shell)
   210	            .arg("-e")
   211	            .arg("-c")
   212	            .arg(&script)
   213	            .output()?
   214	    } else {
   215	        // Windows: write a temp .bat with error-abort after each command.
   216	        let bat_path = std::env::temp_dir().join(format!("thru_exec_{}.bat", std::process::id()));
   217	        let mut bat = String::from("@echo off\r\n");
   218	        for line in script.lines() {
   219	            if !line.trim().is_empty() {
   220	                bat.push_str(line);
   221	                bat.push_str("\r\nif errorlevel 1 exit\r\n");
   222	            }
   223	        }
   224	        fs::write(&bat_path, bat)?;
   225	        let out = Command::new("cmd.exe")
   226	            .arg("/Q")
   227	            .arg("/C")
   228	            .arg(&bat_path)
   229	            .output()?;
   230	        let _ = fs::remove_file(&bat_path);
   231	        out
   232	    };
   233	
   234	    // Stream stdout then stderr as output frames.
   235	    if !output.stdout.is_empty() {
   236	        write_frame(conn, &output.stdout)?;
   237	    }
   238	    if !output.stderr.is_empty() {
   239	        write_frame(conn, &output.stderr)?;
   240	    }
   241	
   242	    let code = output.status.code().unwrap_or(1);
   243	    // Final frame: i32 exit code (big-endian).
   244	    write_frame(conn, &(code as i32).to_be_bytes())?;
   245	    write_frame(conn, &[]) // terminator
   246	}

   250	/// Execute a script on an existing connection; returns (combined_output, exit_code).
   251	pub fn exec_script_on_conn(conn: &mut Box<dyn Connection>, script: &str) -> io::Result<(String, i32)> {
   252	    let mut frame = vec![OP_EXEC_SCRIPT];
   253	    frame.extend_from_slice(&(script.len() as u32).to_be_bytes());
   254	    frame.extend_from_slice(script.as_bytes());
   255	    write_frame(conn, &frame)?;
   256	
   257	    let mut frames: Vec<Vec<u8>> = Vec::new();
   258	    loop {
   259	        let f = read_frame(conn)?;
   260	        if f.is_empty() { break; }
   261	        frames.push(f);
   262	    }
   263	
   264	    let exit_code = if let Some(last) = frames.last() {
   265	        if last.len() == 4 {
   266	            i32::from_be_bytes([last[0], last[1], last[2], last[3]])
   267	        } else { 1 }
   268	    } else { 1 };
   269	
   270	    let mut output = Vec::new();
   271	    for f in &frames[..frames.len().saturating_sub(1)] {
   272	        output.extend_from_slice(f);
   273	    }
   274	    Ok((String::from_utf8_lossy(&output).to_string(), exit_code))
   275	}
